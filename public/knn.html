function sq_euclidean(a, b)
    return vec(sum((a .- b) .^ 2, dims = 1))
end

function knn(dist, train_x, train_y, p, k = 5)
    nearest_idx = partialsortperm(dist(p, train_x), 1:k)
    nearest_labels = train_y[nearest_idx]
    return mode(nearest_labels)
end

function predict(dist, train_x, train_y, test_x, k = 5)
    return [ knn(dist, train_x, train_y, p, k) for p in eachcol(test_x) ]
end

function accuracy(y_preds, y_true)
    return sum(y_preds .== y_true) / length(y_true)
end

function confusion_matrix(y_preds, y_true)
    output_size = length(unique(y_true))
    cm = zeros(output_size,output_size)
    for (y_p, y_t) in zip(y_preds, y_true)
        cm[y_p, y_t] += 1
    end
    return cm
end
