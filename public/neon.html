 9function csv_to_dict(path) {
10    const csv = fs.readFileSync(path,'utf-8');
11    const rows = csv.split(/\r?\n/)
12        .map(
13            r => r
14                .match(/(".*?"|[^",]+)(?=\s*,|\s*$)/g)
15                .map(cell => cell.replace(/^"|"$/, ''))
16        )
17    const header = rows.shift()
18    const data = rows.map(r => Object.fromEntries(
19        header.map((label,i) => [label, r[i]])
20    ))
21    return data
22}
