Sending SMS over an OAuth2 API

 1const nodemailer = require('nodemailer');
 2const { google } = require('googleapis');
 3const dotenv = require('dotenv').config();
 4
 5const CLIENT_ID = process.env.CLIENT_ID;
 6const CLIENT_SECRET = process.env.CLIENT_SECRET;
 7const REFRESH_TOKEN = process.env.REFRESH_TOKEN;
 8const REDIRECT_URI = 'https://developers.google.com/oauthplayground';
 9
10const TO_NUMBER = '6508129777@vtext.com';
11const TO_MESSAGE = 'Welcome to the groovy'
12
13const oAuth2Client = new google.auth.OAuth2(
14  CLIENT_ID, CLIENT_SECRET, REDIRECT_URI
15);
16oAuth2Client.setCredentials({ refresh_token: REFRESH_TOKEN });
17
18async function sendSMS() {
19  try {
20    const accessToken = await oAuth2Client.getAccessToken();
21
22    const transporter = nodemailer.createTransport({
23      service: 'gmail',
24      auth: {
25        type: 'OAuth2',
26        user: 'gordon@liang.ca',
27        clientId: CLIENT_ID,
28        clientSecret: CLIENT_SECRET,
29        refreshToken: REFRESH_TOKEN,
30        accessToken: accessToken.token,
31      },
32    });
33
34    const result = await transporter.sendMail({
35      from: 'your.email@gmail.com',
36      to: TO_NUMBER,
37      text: TO_MESSAGE,
38    });
39
40    console.log('SMS sent:', result.response);
41  } catch (error) {
42    console.error('Error sending SMS:', error);
43  }
44}

46sendSMS();
