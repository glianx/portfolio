function ∇aoL(y_hat, y)
    return (y_hat .- y) ./ length(y)
end

function ∇woL(zh, y_hat, y)
    return ∇aoL(y_hat, y) * zh'
end

function ∇boL(y_hat, y)
    return ∇aoL(y_hat, y)
end

function ∇zhL(y_hat, y, Wo)
    return Wo' * ∇aoL(y_hat, y)
end

function ∇ahL(ah, y_hat, y, Wo)
    return ∇zhL(y_hat, y, Wo) .* ifelse.(ah .> 0, 1.0, 0.01)
end

function ∇whL(ah, y_hat, y, Wo, input)
    return ∇ahL(ah, y_hat, y, Wo) * input'
end

function ∇bhL(ah, y_hat, y, Wo)
    return ∇ahL(ah, y_hat, y, Wo)
end

function train_model(α, k, rng)
    Wh = randn(Float64, (LATENT_SIZE, INPUT_SIZE)) .* sqrt(2/INPUT_SIZE)
    Wo = randn(Float64, (OUTPUT_SIZE, LATENT_SIZE)) .* sqrt(2/LATENT_SIZE)
    bh = zeros(Float64, LATENT_SIZE)
    bo = zeros(Float64, OUTPUT_SIZE)

    frames = Matrix{Gray{Float64}}[]

    for epoch in 1:k
        for i in rng
            input = train_x[:,:,i]
            input = flatten(input)

            ah = pre_activation(input, Wh, bh)
            zh = post_activation(leaky_relu, ah)

            ao = pre_activation(zh, Wo, bo)

            # AUTOENCODER
            y     = input
            y_hat = ao

            Wh -= α * ∇whL(ah, y_hat, y, Wo, input)
            bh -= α * ∇bhL(ah, y_hat, y, Wo)

            Wo -= α * ∇woL(zh,y_hat,y)
            bo -= α * ∇boL(y_hat,y)
            
        end

        temp_model = Model(Wh, Wo, bh, bo)
        push!(frames, Gray.(reconstruct(temp_model, train_x[:,:,5])))
    end

    grid = mosaic(frames...; nrow=4, npad=2, fillvalue=Gray(1.0))
    display(repeat(grid, inner=(4,4)))

    return Wh,Wo,bh,bo
end

function reconstruct(model, input)
    input_shape = size(input)
    input = flatten(input)

    ah = pre_activation(input, model.Wh, model.bh)
    zh = post_activation(leaky_relu, ah)

    ao    = pre_activation(zh, model.Wo, model.bo)
    y_hat = ao

    return reshape(y_hat, input_shape)'
end
