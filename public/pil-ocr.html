 1import os

 6def get_ocr_pages(pdf_path, img_dir, dpi = 200, pdf_name = 'pdf'):
 7    pages = []
 8
 9    imgs = pdf2image.convert_from_path(pdf_path, dpi)
10    for i, img in enumerate(imgs):
11        img = img.convert('L')
12        img = ImageEnhance.Contrast(img).enhance(2)
13        img.save(os.path.join(img_dir, f'{pdf_name}-page-{i+1}.png'))
14
15        page = pytesseract.image_to_string(img)
16        pages.append(page)
17
18    return pages
