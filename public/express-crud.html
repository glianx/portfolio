CRUD over HTTP with Express and SQLite

 1import express from 'express';
 2import Database from 'better-sqlite3';
 3
 4const db = new Database('colors.db');
 5db.prepare(
 6    `CREATE TABLE IF NOT EXISTS colors (
 7        id INTEGER PRIMARY KEY, 
 8        name TEXT, 
 9        shade NUMBER, 
10        dark BOOLEAN
11    )`
12).run();
13
14const app = express();
15app.use(express.json())
16app.get('/', (req, res) => {
17  res.sendFile(process.cwd() + '/main.html');
18});
19
20app.get('/api/colors', (req, res) => {
21    const stmt = db.prepare("SELECT * FROM colors");
22    const result = stmt.all();
23    res.json(result);
24});
25
26app.get('/api/colors/:id', (req, res) => {
27    const id = req.params.id;
28    if (!id) return res.status(400).json({ error: 'Missing id' });
29
30    const stmt = db.prepare("SELECT * FROM colors WHERE id = :id");
31    const result = stmt.get({ id });
32    res.json(result);
33});
34
35app.post('/api/colors', (req, res) => {
36  const name  = req.body.name;
37  const shade = req.body.shade;
38  const dark  = req.body.dark ? 1 : 0;
39  if (!name) return res.status(400).json({ error: 'Missing name' });
40
41  const stmt = db.prepare("INSERT INTO colors (name, shade, dark) VALUES (:name, :shade, :dark)");
42  const result = stmt.run({ name, shade, dark });
43  res.json({ id: result.lastInsertRowid, name: name, shade: shade, dark: dark });
44});
45
46app.put('/api/colors/:id', (req, res) => {
47    const id = req.params.id;
48    const name  = req.body.name;
49    const shade = req.body.shade;
50    const dark  = req.body.dark ? 1 : 0;
51    const stmt  = db.prepare("UPDATE colors SET name = :name, shade = :shade, dark = :dark WHERE id = :id");
52    const result = stmt.run({ id, name, shade, dark });
53    res.json(result);
54});
55
56app.patch('/api/colors/:id', (req, res) => {
57    const id = req.params.id;
58    const allowedFields = ['name', 'shade', 'dark'];
59    
60    const updates = {};
61    if (req.body.name) updates.name = req.body.name;
62    if (req.body.shade) updates.shade = req.body.shade;
63    if ('dark' in req.body) updates.dark = req.body.dark ? 1 : 0; 
64
65    const fields = Object.keys(updates)
66        .map((key) => `${key} = :${key}`)
67        .join(',');
68    const stmt = db.prepare(`UPDATE colors SET ${fields} WHERE id = :id`);
69    const result = stmt.run({ ...updates, id });
70    res.json(result);
71})

73app.delete('/api/colors', (req, res) => {
74    const stmt = db.prepare("DELETE FROM colors");
75    const result = stmt.run();
76    res.json(result);
77});

79app.delete('/api/colors/:id', (req, res) => {
