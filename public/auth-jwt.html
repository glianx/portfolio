Password hashing with bcrypt and JWT sessions

server.ts · 81 lines

 1import express from "express";
 2import bcrypt from "bcrypt";
 3import jwt from "jsonwebtoken";
 4
 5// Define user interface
 6interface User {
 7  username: string;
 8  hash: string;
 9}
10
11// In-memory user database
12const users: User[] = [];
13
14// Create express application
15const app = express();
16
17// Middleware for parsing JSON bodies
18app.use(express.json());
19
20// Register endpoint
21app.post("/register", async (req, res) => {
22  try {
23    const { username, password } = req.body;
24    const hash = await bcrypt.hash(password, 10);
25    users.push({ username, hash });
26    res.status(201).send("Registered");
27  } catch (error) {
28    res.status(500).send("Error registering user");
29  }
30});
31
32// Login endpoint
33app.post("/login", async (req, res) => {
34  try {
35    const { username, password } = req.body;
36    const user = users.find(u => u.username === username);
37    
38    if (!user) {
39      res.status(401).send("Unauthorized");
40      return;
41    }
42    
43    const match = await bcrypt.compare(password, user.hash);
44    if (!match) {
45      res.status(401).send("Unauthorized");
46      return;
47    }
48    
49    const token = jwt.sign({ username }, "secret", { expiresIn: "1h" });
50    res.json({ token });
51  } catch (error) {
52    res.status(500).send("Error logging in");
53  }
54});
55
56// Protected endpoint
57app.get("/protected", (req, res) => {
58  try {
59    const auth = req.headers.authorization;
60    
61    if (!auth) {
62      res.status(401).send("No token");
63      return;
64    }
65
66    const token = auth.split(" ")[1];
67    const decoded = jwt.verify(token, "secret") as { username: string };
68    res.send(`Hello ${decoded.username}`);
69  } catch (error) {
70    res.status(403).send("Invalid token");
71  }
72});
73
74// Error handling middleware
75app.use((err: any, req: express.Request, res: express.Response, next: express.NextFunction) => {
76  console.error(err.stack);
77  res.status(500).send('Something broke!');
78});
79
80// Start server
81app.listen(3000, () => console.log("Server running on http://localhost:3000"));
