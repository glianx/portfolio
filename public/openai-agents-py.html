 1from agents import Agent, Runner, InputGuardrail, GuardrailFunctionOutput
 2from agents.exceptions import InputGuardrailTripwireTriggered
 3from pydantic import BaseModel
 4import asyncio
 5
 6class HomeworkOutput(BaseModel):
 7    is_homework: bool
 8    reasoning: str
 9
10history_haiku_agent = Agent(
11    name = 'history haiku',
12    handoff_description = 'specialist tutor for answering history queries',
13    instructions = 'you are a history haiku tutor. answer history queries with a haiku.'
14)
15
16math_joker_agent = Agent(
17    name = 'math joker',
18    handoff_description = 'specialist tutor for answering math queries',
19    instructions = 'you are a math joker tutor. answer math queries with a joke.'
20)
21
22guardrail_agent = Agent(
23    name = 'guardrail agent',
24    instructions = 'check if user is asking about homework',
25    output_type = HomeworkOutput
26)
27
28async def homework_guardrail(ctx, agent, input_data):
29    result = await Runner.run(guardrail_agent, input_data, context = ctx.context)
30    final_output = result.final_output_as(HomeworkOutput)
31    return GuardrailFunctionOutput(
32        output_info = final_output,
33        tripwire_triggered = not final_output.is_homework
34    )
35
36triage_agent = Agent(
37    name = 'triage agent',
38    instructions = "you determine which agent to use based on user's homework question",
39    handoffs = [history_haiku_agent, math_joker_agent],
40    input_guardrails = [
41        InputGuardrail(guardrail_function = homework_guardrail)
42    ]
43)
44
45async def main():
46    try:
47        result = await Runner.run(triage_agent, 'who was first US president?')
48        print(result.final_output)
49    except InputGuardrailTripwireTriggered as e:
50        print("Guardrail blocked this input: ", e)
51
52    try:
